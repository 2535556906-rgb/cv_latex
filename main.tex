% ===== projects =====
\cvsection{Selected Projects}

\cventry{Sensorless FOC Drive for Electric Vehicles}{Dec 2025 -- Jun 2026}%
  {Bachelor's thesis --- Shanghai University}
\begin{cvitems}
  \item Built a sensorless surface-mounted PMSM FOC drive on STM32G431 with dual
        current/speed loops and SVPWM, running stably without Hall sensors.
  \item Estimated rotor flux with a nonlinear observer and extracted angle/speed
        through a PLL; identified inductance and flux online via $\alpha$-axis
        high-frequency injection and DFT, improving accuracy under heavy load and
        uphill conditions.
  \item Designed the power stage (FD6288T gate driver, three-phase bridge,
        three-shunt sensing, multi-stage DC-DC) and ported the validated firmware
        to Keil.
\end{cvitems}

\cventry{Automatic Needle-Insertion Machine for Pre-filled Syringes}{Dec 2024 -- Jun 2025}%
  {Industry project --- Technical Lead}
\begin{cvitems}
  \item Developed low-level motion control for a 3-axis servo system: coordinate
        solving, linear interpolation, acceleration/deceleration planning and
        G-code parsing.
  \item Led a 16-person team and owned the overall technical solution and delivery
        schedule.
\end{cvitems}

\cventry{Self-Balancing Line-Tracking Motorcycle}{Dec 2023 -- Aug 2024}%
  {Smart Car Competition --- Team captain}
\begin{cvitems}
  \item Designed a cascade PID controller (angular rate -- angle -- tracking error)
        for static balance and high-speed cornering.
  \item Implemented 8-neighbourhood edge scanning for track recognition; fused
        gyroscope and accelerometer data with an EKF (drift $<0.5^{\circ}$ over 10 min).
\end{cvitems}

\cventry{Vehicle-Mounted Auto-Aiming Laser Shooting Device}{Aug 2025}%
  {National Electronics Design Contest --- Team captain}
\begin{cvitems}
  \item Decoupled gimbal and chassis with cascade PID, keeping the aim stable while
        the vehicle was moving.
  \item Detected target corners, recovered the bullseye by inverse perspective
        mapping and locked a laser onto the target in motion.
\end{cvitems}

\cventry{FPGA-Based Mini-LED Local-Dimming Display}{Sep 2024 -- Dec 2024}%
